# --- MPU6050 Driver ---
def init_mpu6050():
    try:
        # Wake up sensor
        i2c.writeto_mem(MPU_ADDR, 0x6B, b'\x00')
        time.sleep_ms(100)
        print("MPU6050 Initialized")
    except Exception as e:
        print("MPU6050 Error:", e)

def read_raw_sensor():
    """
    Returns (ax, ay, az, gx, gy, gz)
    Units: G-force and Degrees/sec
    """
    try:
        data = i2c.readfrom_mem(MPU_ADDR, 0x3B, 14)
        vals = ustruct.unpack('>hhhhhhh', data)
        
        # Accel: 16384 LSB/g
        ax = vals[0] / 16384.0
        ay = vals[1] / 16384.0
        az = vals[2] / 16384.0
        
        # Gyro: 131 LSB/dps
        gx = vals[4] / 131.0
        gy = vals[5] / 131.0
        gz = vals[6] / 131.0
        
        return ax, ay, az, gx, gy, gz
    except:
        return 0,0,0,0,0,0

# 4. FEATURE EXTRACTION
# ---------------------------------------------------------
def calculate_features(buffer):
    """
    Converts 50 raw samples -> 56 Statistical Features
    Must match the 'extract_features' function in Colab EXACTLY.
    """
    features = []
    
    # We have 8 columns (sensors) in the buffer
    # buffer[row][col]
    
    for col_idx in range(8):
        # Extract one signal column (e.g., all acc_x values)
        col_data = [row[col_idx] for row in buffer]
        
        # 1. Mean
        avg = sum(col_data) / len(col_data)
        
        # 2. Std Dev
        var = sum((x - avg) ** 2 for x in col_data) / len(col_data)
        std = math.sqrt(var)

        # 3. Max & Min
        maxim = max(col_data)
        minim = min(col_data)
        
        # 4. Range
        rng = maxim - minim
        
        # 5. Energy (Sum of Squares / N)
        energy = sum(x**2 for x in col_data) / len(col_data)
        
        # 6. Jerk (Sum of absolute differences)
        # sum(|x[i] - x[i-1]|)
        jerk = 0
        for i in range(1, len(col_data)):
            jerk += abs(col_data[i] - col_data[i-1])
            
        # Add to feature vector in EXACT order used in training
        # [mean, std, max, min, range, energy, jerk]
        features.extend([avg, std, maxim, minim, rng, energy, jerk])
        
    return features
